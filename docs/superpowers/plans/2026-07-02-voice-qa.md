# Voice Q&A Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a kid ask a spoken question during a module and get a spoken, topic-scoped answer — mic capture → speech-to-text → Claude API → text-to-speech — with a safe system prompt, a visible transcript, and an offline/error fallback to pre-recorded suggested answers.

**Architecture:** The Claude API call and the API key live in the **Electron main process**, never in the renderer — the renderer calls `window.api.askClaude(question, moduleId)` over IPC, so the key is never exposed to devtools/renderer JS and the request isn't subject to browser CORS. All browser-only APIs (`SpeechRecognition`, `speechSynthesis`) are wrapped behind small dependency-injected interfaces so the orchestration logic (`useVoiceQA`) is fully unit-testable without a real browser engine. The redirect behavior for off-topic questions is driven by a `[REDIRECT]` marker the system prompt asks Claude to emit, detected and stripped in `claudeService.ts`.

**Tech Stack:** Anthropic Claude API (`claude-haiku-4-5`) called from Node (Electron main), Electron IPC (`ipcMain.handle` / `contextBridge`), Web Speech API (`SpeechRecognition`, `speechSynthesis`) wrapped for testability, React, Vitest.

// electron-app/src/main/config.ts
import { promises as fs } from 'node:fs'
import path from 'node:path'

function configFilePath(configDir: string): string {
  return path.join(configDir, 'planetary-movement-config.json')
}

export async function readApiKey(configDir: string): Promise<string | null> {
  try {
    const raw = await fs.readFile(configFilePath(configDir), 'utf-8')
    const parsed = JSON.parse(raw) as { claudeApiKey?: unknown }
    return typeof parsed.claudeApiKey === 'string' ? parsed.claudeApiKey : null
  } catch (err) {
    if ((err as NodeJS.ErrnoException).code === 'ENOENT') return null
    throw err
  }
}

export async function writeApiKey(configDir: string, apiKey: string): Promise<void> {
  await fs.mkdir(configDir, { recursive: true })
  await fs.writeFile(
    configFilePath(configDir),
    JSON.stringify({ claudeApiKey: apiKey }, null, 2),
    'utf-8'
  )
}
```

// electron-app/src/main/claudeService.ts
export function buildSystemPrompt(moduleTitle: string, moduleSummary: string): string {
  return [
    `You are a friendly science guide inside an educational app for an 11-13 year old, currently exploring the "${moduleTitle}" module: ${moduleSummary}`,
    'Answer only questions about Earth and space science related to this app (planetary motion, day/night, seasons, climate, weather, plate tectonics, tsunamis, tides).',
    'Keep answers short: 2 to 4 sentences, spoken-friendly, no markdown, no lists, no headings — this will be read aloud.',
    'Use vocabulary appropriate for an 11-13 year old, but do not oversimplify real terms like axis, orbit, tilt, or tectonic plates.',
    'If the question is unrelated to Earth/space science, or inappropriate, respond with exactly this format: [REDIRECT] followed by a short friendly redirect back to the current module topic.'
  ].join(' ')
}

export interface AskClaudeParams {
  apiKey: string
  question: string
  moduleTitle: string
  moduleSummary: string
  fetchImpl?: typeof fetch
}

export interface AskClaudeResult {
  answer: string
  redirected: boolean
}

export async function askClaude({
  apiKey,
  question,
  moduleTitle,
  moduleSummary,
  fetchImpl = fetch
}: AskClaudeParams): Promise<AskClaudeResult> {
  const response = await fetchImpl('https://api.anthropic.com/v1/messages', {
    method: 'POST',
    headers: {
      'x-api-key': apiKey,
      'anthropic-version': '2023-06-01',
      'content-type': 'application/json'
    },
    body: JSON.stringify({
      model: 'claude-haiku-4-5',
      max_tokens: 200,
      system: buildSystemPrompt(moduleTitle, moduleSummary),
      messages: [{ role: 'user', content: question }]
    })
  })

  if (!response.ok) {
    const detail = await response.text()
    throw new Error(`Claude API request failed (${response.status}): ${detail}`)
  }

  const data = (await response.json()) as { content: Array<{ type: string; text: string }> }
  const rawText = data.content?.[0]?.text ?? ''

  if (rawText.startsWith('[REDIRECT]')) {
    return { answer: rawText.replace('[REDIRECT]', '').trim(), redirected: true }
  }

  return { answer: rawText.trim(), redirected: false }
}
```

// electron-app/src/renderer/src/voice/speechRecognizer.ts
interface SpeechRecognitionResultLike {
  transcript: string
}

interface SpeechRecognitionEventLike {
  results: SpeechRecognitionResultLike[][]
}

interface SpeechRecognitionErrorEventLike {
  error: string
}

interface SpeechRecognitionLike {
  lang: string
  continuous: boolean
  interimResults: boolean
  start: () => void
  stop: () => void
  onresult: ((event: SpeechRecognitionEventLike) => void) | null
  onerror: ((event: SpeechRecognitionErrorEventLike) => void) | null
  onend: (() => void) | null
}

type SpeechRecognitionCtor = new () => SpeechRecognitionLike

export interface SpeechRecognizerHandlers {
  onResult: (transcript: string) => void
  onError: (message: string) => void
  onEnd: () => void
}

export interface SpeechRecognizerController {
  start: () => void
  stop: () => void
}

function getGlobalSpeechRecognitionCtor(): SpeechRecognitionCtor | undefined {
  const w = window as unknown as {
    SpeechRecognition?: SpeechRecognitionCtor
    webkitSpeechRecognition?: SpeechRecognitionCtor
  }
  return w.SpeechRecognition ?? w.webkitSpeechRecognition
}

export function createSpeechRecognizer(
  handlers: SpeechRecognizerHandlers,
  ctor: SpeechRecognitionCtor | undefined = getGlobalSpeechRecognitionCtor()
): SpeechRecognizerController {
  if (!ctor) {
    throw new Error('Speech recognition is not supported in this environment')
  }

  const recognition = new ctor()
  recognition.lang = 'en-US'
  recognition.continuous = false
  recognition.interimResults = false

  recognition.onresult = (event) => {
    const transcript = event.results[0]?.[0]?.transcript ?? ''
    handlers.onResult(transcript)
  }
  recognition.onerror = (event) => {
    handlers.onError(event.error)
  }
  recognition.onend = () => {
    handlers.onEnd()
  }

  return {
    start: () => recognition.start(),
    stop: () => recognition.stop()
  }
}
```

// electron-app/src/renderer/src/voice/speechSynthesizer.ts
interface SpeechSynthesisUtteranceLike {
  text: string
  onend: (() => void) | null
  onerror: ((event: unknown) => void) | null
}

interface SpeechSynthesizerLike {
  speak: (utterance: SpeechSynthesisUtteranceLike) => void
  cancel: () => void
}

type UtteranceCtor = new (text: string) => SpeechSynthesisUtteranceLike

export function speak(
  text: string,
  synth: SpeechSynthesizerLike = window.speechSynthesis as unknown as SpeechSynthesizerLike,
  UtteranceCtor: UtteranceCtor = window.SpeechSynthesisUtterance as unknown as UtteranceCtor
): Promise<void> {
  return new Promise((resolve, reject) => {
    const utterance = new UtteranceCtor(text)
    utterance.onend = () => resolve()
    utterance.onerror = () => reject(new Error('Speech synthesis failed'))
    synth.speak(utterance)
  })
}

// electron-app/src/renderer/src/voice/useVoiceQA.ts
import { useCallback, useRef, useState } from 'react'
import type { SpeechRecognizerController, SpeechRecognizerHandlers } from './speechRecognizer'

export interface TranscriptEntry {
  id: string
  question: string
  answer: string
}

export interface VoiceQAState {
  isListening: boolean
  isThinking: boolean
  error: string | null
  transcript: TranscriptEntry[]
}

export interface VoiceQADeps {
  askClaude: (question: string, moduleId: string) => Promise<{ answer: string; redirected: boolean }>
  speak: (text: string) => Promise<void>
  createRecognizer: (handlers: SpeechRecognizerHandlers) => SpeechRecognizerController
}

const initialState: VoiceQAState = {
  isListening: false,
  isThinking: false,
  error: null,
  transcript: []
}

export function useVoiceQA(moduleId: string, deps: VoiceQADeps) {
  const [state, setState] = useState<VoiceQAState>(initialState)
  const recognizerRef = useRef<SpeechRecognizerController | null>(null)

  const askAndRespond = useCallback(
    async (question: string) => {
      setState((s) => ({ ...s, isThinking: true, error: null }))
      try {
        const { answer } = await deps.askClaude(question, moduleId)
        setState((s) => ({
          ...s,
          isThinking: false,
          transcript: [...s.transcript, { id: crypto.randomUUID(), question, answer }]
        }))
        await deps.speak(answer)
      } catch (err) {
        setState((s) => ({ ...s, isThinking: false, isListening: false, error: (err as Error).message }))
      }
    },
    [deps, moduleId]
  )

  const startListening = useCallback(() => {
    setState((s) => ({ ...s, isListening: true, error: null }))
    recognizerRef.current = deps.createRecognizer({
      onResult: (transcript) => {
        void askAndRespond(transcript)
      },
      onError: (message) => {
        setState((s) => ({ ...s, isListening: false, error: message }))
      },
      onEnd: () => {
        setState((s) => ({ ...s, isListening: false }))
      }
    })
    recognizerRef.current.start()
  }, [deps, askAndRespond])

  const stopListening = useCallback(() => {
    recognizerRef.current?.stop()
  }, [])

  const addFallbackEntry = useCallback((question: string, answer: string) => {
    setState((s) => ({
      ...s,
      transcript: [...s.transcript, { id: crypto.randomUUID(), question, answer }]
    }))
  }, [])

  return { state, startListening, stopListening, askAndRespond, addFallbackEntry }
}
```

// electron-app/src/main/ipcHandlers.ts
import { app, ipcMain } from 'electron'
import { readApiKey, writeApiKey } from './config'
import { askClaude } from './claudeService'
import { getModuleById } from '../renderer/src/data/modules'

export function registerIpcHandlers(): void {
  const configDir = app.getPath('userData')

  ipcMain.handle('get-api-key-status', async () => {
    const key = await readApiKey(configDir)
    return { hasKey: Boolean(key) }
  })

  ipcMain.handle('set-api-key', async (_event, apiKey: string) => {
    await writeApiKey(configDir, apiKey)
  })

  ipcMain.handle('ask-claude', async (_event, question: string, moduleId: string) => {
    const apiKey = await readApiKey(configDir)
    if (!apiKey) {
      throw new Error('No Claude API key configured')
    }
    const module = getModuleById(moduleId)
    if (!module) {
      throw new Error(`Unknown module: ${moduleId}`)
    }
    return askClaude({
      apiKey,
      question,
      moduleTitle: module.title,
      moduleSummary: module.summary
    })
  })
}
