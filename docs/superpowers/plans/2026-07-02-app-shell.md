> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up a working Electron + React desktop window with module navigation, so later plans (Unity scene, narration, voice Q&A) have somewhere to embed into.

**Architecture:** `electron-vite` scaffold (Electron main/preload/renderer split) with a React + TypeScript renderer. Module list is static data; nav renders it and shows a placeholder view per module. No 3D, no voice, no audio yet — those are separate plans.

**Tech Stack:** Electron, React 18, TypeScript, electron-vite, Vitest, @testing-library/react

// electron-app/src/renderer/src/components/ModuleNav.tsx
import { MODULES } from '../data/modules'

interface ModuleNavProps {
  activeModuleId: string
  onSelect: (moduleId: string) => void
}

export function ModuleNav({ activeModuleId, onSelect }: ModuleNavProps): JSX.Element {
  return (
    <nav aria-label="Curriculum modules">
      <ul>
        {MODULES.map((module) => (
          <li key={module.id}>
            <button
              type="button"
              disabled={!module.implemented}
              aria-current={module.id === activeModuleId ? 'true' : undefined}
              onClick={() => onSelect(module.id)}
            >
              {module.title}
            </button>
          </li>
        ))}
      </ul>
    </nav>
  )
}
```

// electron-app/src/renderer/src/components/ModuleView.tsx
import { getModuleById } from '../data/modules'

interface ModuleViewProps {
  moduleId: string
}

export function ModuleView({ moduleId }: ModuleViewProps): JSX.Element {
  const module = getModuleById(moduleId)

  if (!module) {
    return <p>Module not found.</p>
  }

  return (
    <section>
      <h1>{module.title}</h1>
      <p>{module.summary}</p>
      {!module.implemented && <p>Coming soon.</p>}
      {module.implemented && (
        <div data-testid="module-scene-slot">
          {/* Unity scene, narration, and voice Q&A mount here — added in later plans */}
        </div>
      )}
    </section>
  )
}
```
